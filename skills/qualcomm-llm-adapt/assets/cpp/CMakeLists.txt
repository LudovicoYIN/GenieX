cmake_minimum_required(VERSION 3.17)
project(geniex_chat_demo CXX)
set(CMAKE_CXX_STANDARD 17)
set(GENIEX_SDK_DIR "" CACHE PATH "SDK root containing include/geniex.h and lib/libgeniex.so")
if(NOT EXISTS "${GENIEX_SDK_DIR}/include/geniex.h" OR NOT EXISTS "${GENIEX_SDK_DIR}/lib/libgeniex.so")
    message(FATAL_ERROR "Set GENIEX_SDK_DIR to a matching deployed SDK")
endif()
add_library(geniex SHARED IMPORTED)
set_target_properties(geniex PROPERTIES IMPORTED_LOCATION "${GENIEX_SDK_DIR}/lib/libgeniex.so")
add_executable(geniex-chat chat.cpp)
target_include_directories(geniex-chat PRIVATE "${GENIEX_SDK_DIR}/include")
target_link_libraries(geniex-chat PRIVATE geniex)
add_executable(argmax-neon argmax_neon.cpp)
